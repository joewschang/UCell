L1860 CREATE OR REPLACE FUNCTION organization.fn_validate_first_third_left_rule()
L1861 RETURNS trigger
L1862 LANGUAGE plpgsql
L1863 AS $$
L1864 DECLARE
L1865   v_sponsor uuid;
L1866   v_seq integer;
L1867   v_requires_left boolean;
L1868   v_ok boolean;
L1869 BEGIN
L1870   SELECT sponsor_qualification_id, sponsor_sequence_no
L1871   INTO v_sponsor, v_seq
L1872   FROM organization.sponsor_relationship
L1873   WHERE child_qualification_id = NEW.qualification_id
L1874     AND effective_to IS NULL
L1875   ORDER BY effective_from DESC
L1876   LIMIT 1;
L1877 
L1878   IF v_sponsor IS NULL THEN
L1879     RETURN NEW;
L1880   END IF;
L1881 
L1882   v_requires_left := v_seq IN (1,3);
L1883 
L1884   IF v_requires_left THEN
L1885     -- Candidate is valid if its parent is sponsor's LEFT child or anywhere below it.
L1886     IF NEW.parent_qualification_id = v_sponsor AND NEW.side = 'LEFT' THEN
L1887       v_ok := true;
L1888     ELSE
L1889       v_ok := organization.fn_is_in_binary_subtree(v_sponsor, NEW.parent_qualification_id, 'LEFT');
L1890     END IF;
L1891 
L1892     IF NOT coalesce(v_ok,false) THEN
L1893       RAISE EXCEPTION 'BINARY_LEFT_SUBTREE_REQUIRED: sponsor sequence % must be placed in sponsor left subtree', v_seq
L1894         USING ERRCODE='23514';
L1895     END IF;
L1896   END IF;
L1897 
L1898   RETURN NEW;
L1899 END;
L1900 $$;
